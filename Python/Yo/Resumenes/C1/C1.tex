\section{2. Manipulación de Dígitos (Sin usar str())}
\textbf{Dado $N \ge 0$ (REGLA: NUNCA usar str(N) si lo prohíben):}
\begin{itemize}
    \item \textbf{Último dígito (der):} \texttt{n \% 10} \quad \textbf{Quitar der:} \texttt{n // 10}
    \item \textbf{Último par (base 100):} \texttt{n \% 100} \quad \textbf{Quitar par:} \texttt{n // 100}
    \item \textbf{Primeros 2 dígitos (izq):} \texttt{n // (10 ** (digitos(n) - 2))}
    \item \textbf{Quitar primeros 2 dígitos:} \texttt{n \% (10 ** (digitos(n) - 2))}
    \item \textbf{Quitar ambos extremos (2 dígitos c/u):} \\
    \texttt{nuevo = (n \% (10**(largo - 2))) // 100}
    \item \textbf{Pegar número a la izquierda:} \texttt{izq * (10**digitos(der)) + der}
    \item \textbf{Pegar número a la derecha:} \texttt{juego * 100 + nuevaFicha}
\end{itemize}

\section{3. Primos y Divisores (Clase 8)}
\subsection{Verificar si un número es Primo ($O(\sqrt{n})$)}
\begin{verbatim}
# esPrimo: int -> bool
def esPrimo(n):
    if n == 2: return True
    elif n % 2 == 0 or n < 2: return False
    return esPrimoRec(n, 3)

# esPrimoRec: int int -> bool
def esPrimoRec(n, div=3):
    if div ** 2 > n: return True
    elif n % div == 0: return False
    else: return esPrimoRec(n, div + 2)
\end{verbatim}

\subsection{Generar String con los primeros $n$ Primos}
\begin{verbatim}
# nPrimos: int -> str
def nPrimos(n):
    if n == 0: return ''
    elif n == 1: return '2'
    else: return '2' + nPrimosRec(n - 1, 3)

# nPrimosRec: int int -> str
def nPrimosRec(n, valor):
    if n == 0: return ''
    elif esPrimo(valor):
        return ' ' + str(valor) + nPrimosRec(n - 1, valor + 2)
    else: return nPrimosRec(n, valor + 2)
\end{verbatim}

\subsection{Factores Primos de un Número}
\begin{verbatim}
# menorDivisor: int int -> int
def menorDivisor(n, div):
    if n % div == 0: return div
    return menorDivisor(n, div + 1)

# factoresPrimos: int -> str
def factoresPrimos(n):
    if esPrimo(n): return str(n)
    else:
        div = menorDivisor(n, 2)
        return str(div) + ' ' + str(factoresPrimos(n // div))
\end{verbatim}

\section{4. Raíz Cuadrada: Método de Herón (Clase 7)}
\begin{verbatim}
# cerca: num num num -> bool
def cerca(x, y, eps=0.0001):
    return abs(x - y) < eps

# heronRec: num num num -> num
def heronRec(x, eps, est):
    assert x > 0
    if cerca(est * est, x, eps): return est
    else:
        mejor = (est + (x / est)) / 2.0
        return heronRec(x, eps, mejor)

# heron: num num -> num
def heron(x, eps):
    return heronRec(x, eps, x / 2.0)
\end{verbatim}

\section{5. Torres de Hanoi (Recursión con Strings)}
\begin{verbatim}
# hanoiRec: int str str str -> str
def hanoiRec(n, partida, medio, final):
    if n == 0: return ''
    t1 = hanoiRec(n - 1, partida, final, medio)
    t2 = str(partida) + ' --> ' + str(final) + '\n'
    t3 = hanoiRec(n - 1, medio, partida, final)
    return t1 + t2 + t3
\end{verbatim}

\subsection{Triángulo de Sierpinski}
\begin{verbatim}
def triangulo(largo):
    turtle.forward(largo); turtle.left(120)
    turtle.forward(largo); turtle.left(120)
    turtle.forward(largo); turtle.left(120)

def sierpinski(largo):
    if largo > 4:
        triangulo(largo)
        sierpinski(largo // 2)
        turtle.forward(largo // 2)
        sierpinski(largo // 2)
        turtle.backward(largo // 2); turtle.left(60)
        turtle.forward(largo // 2); turtle.right(60)
        sierpinski(largo // 2)
        turtle.left(60); turtle.backward(largo // 2); turtle.right(60)
\end{verbatim}

\subsection{Copo de Nieve de Koch}
\begin{verbatim}
def koch(size, min_size):
    if size / 3 < min_size:
        turtle.forward(size)
    else:
        koch(size / 3, min_size); turtle.left(60)
        koch(size / 3, min_size); turtle.right(120)
        koch(size / 3, min_size); turtle.left(60)
        koch(size / 3, min_size)
\end{verbatim}

\section{7. Plantillas Recursivas Fundamentales}
\subsection{Contar Dígitos / Invertir / Mayor Dígito}
\begin{verbatim}
def digitos(n):
    if n < 10: return 1
    return 1 + digitos(n // 10)

def invertir(n):
    if n < 10: return n
    return (n % 10)*(10**digitos(n // 10)) + invertir(n // 10)

def maxDigito(n):
    if n < 10: return n
    r = maxDigito(n // 10); u = n % 10
    return u if u > r else r
\end{verbatim}

\subsection{Procedimiento Recursivo (Mostrar / Imprimir)}
\begin{verbatim}
def mostrar(n):
    if n > 0:
        print("Dato:", n % 100)
        mostrar(n // 100)
\end{verbatim}

\section{8. Operadores, Módulos y Trucos Clave}
\begin{itemize}
    \item \textbf{Precedencia Operadores:} \texttt{**} $>$ \texttt{*}, \texttt{/}, \texttt{//}, \texttt{\%} $>$ \texttt{+}, \texttt{-} $>$ Relacionales (\texttt{<, >, ==, !=}) $>$ \texttt{not} $>$ \texttt{and} $>$ \texttt{or}.
    \item \textbf{División Entera vs Real:} \texttt{7 / 2 -> 3.5} (float), \texttt{7 // 2 -> 3} (int), \texttt{7 \% 2 -> 1}.
    \item \textbf{Módulo Math y Random:}
    \begin{verbatim}
import math, random
math.sqrt(x)  | math.pi | round(num, 1)
random.randint(a, b) # Entero en [a, b] inclusivo
random.random()      # Real en [0.0, 1.0)
    \end{verbatim}
    \item \textbf{Redondeo:} \texttt{round(3.956, 1) -> 4.0}, \texttt{round(2/3, 2) -> 0.67}.
    \item \textbf{Input y Conversión limpia:}
    \begin{verbatim}
val = int(input("Ingrese n: "))
txt = input("Nombre? ")
print("Resultado: " + str(val) + " " + txt)
    \end{verbatim}
    \item \textbf{Retorno Booleano Directo:} Escribe siempre \texttt{return (x \% 2 == 0)} en vez de \texttt{if... return True else return False}.
    \item \textbf{Importar módulos de la pregunta:} Si el problema menciona un módulo ficticio (ej. \texttt{domino} o \texttt{musical}), escribe: \texttt{from domino import *} arriba.
\end{itemize}
